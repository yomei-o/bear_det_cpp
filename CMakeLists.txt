cmake_minimum_required(VERSION 3.16)
project(bear_det CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
  add_compile_options(/utf-8 /bigobj /openmp /wd4244 /wd4267 /wd4305 /wd4996)
else()
  find_package(OpenMP)
endif()
add_executable(bdet bdet.cpp)
target_include_directories(bdet PRIVATE pure pure/third_party)
if(NOT MSVC AND OpenMP_CXX_FOUND)
  target_link_libraries(bdet PRIVATE OpenMP::OpenMP_CXX)
endif()
